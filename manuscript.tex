\section{Study 1}\label{study-1}

\textbf{H1 (Difficulty judgments):} \(\beta\) = 0.45, t(911.69) = 22.18, p \textless{} .001.

\textbf{H2 (Competence inferences from a single outcome):} \(\beta\) = 0.86, t(418) = 34.72, p \textless{} .001, Adjusted \(R^2\) = 0.74.

\textbf{H3--H4 (Model accuracy and comparison):}

\begin{itemize}
\tightlist
\item
  Main (Bayesian) Model: \(\beta\) = 0.94, t(418) = 58.29, p \textless{} .001, Adjusted \(R^2\) = 0.89.
\item
  Threshold: Adjusted \(R^2\) = 0.39.
\item
  Anchor: Adjusted \(R^2\) = 0.59.
\item
  BIC: \(BIC_{\text{Bayes}}\) = 16,063.72; \(BIC_{\text{Threshold}}\) = 19,411.92; \(BIC_{\text{Anchor}}\) = 18,259.96.
\end{itemize}

\textbf{Robustness (bottom 30\% of arithmetic performers):}

\begin{itemize}
\tightlist
\item
  Perceived vs.~objective difficulty: \(\beta\) = 0.93, t(303) = 44.51, p \textless{} .001.
\item
  Conditional predictions: \(\beta\) = 0.85, t(448) = 34.03, p \textless{} .001, Adjusted \(R^2\) = 0.72.
\end{itemize}

\section{Study 2}\label{study-2}

\textbf{H1 (Difficulty judgments):} \(\beta\) = 0.39, t(313.06) = 16.85, p \textless{} .001.

\textbf{H2a (Fast accurate answers are competence cues):} b = 0.51, z = 2.15, p = 0.031.

\textbf{H2b (Fast inaccurate answers are competence cues):} b = 0.71, z = 2.63, p = 0.009.

\textbf{H3 (Competence inferences from a single outcome):} \(\beta\) = 0.89, t(98) = 19.47, p \textless{} .001, Adjusted \(R^2\) = 0.79.

\textbf{H4a (Participants treat faster accurate answers as competence cues):} b = 0.17, z = 1.53, p = 0.125.

\textbf{H4b (Participants treat faster inaccurate answers as competence cues):} b = 0.15, z = 1.62, p = 0.104.

\textbf{Robustness (bottom 30\% of arithmetic performers):}

\begin{itemize}
\tightlist
\item
  Perceived vs.~objective difficulty: \(\beta\) = 0.32, t(88.04) = 8.16, p \textless{} .001.
\item
  Conditional predictions: \(\beta\) = 0.88, t(98) = 18.73, p \textless{} .001, Adjusted \(R^2\) = 0.78.
\item
  Trial-level analysis: b = 1.95, z = 46.19, p \textless{} .001.
\end{itemize}
